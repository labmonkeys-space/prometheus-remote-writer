//

// e2e/compose.base.yml:
//
//   IMG=opennms/horizon:36.0.4
//   docker run [-]-rm [-]-entrypoint sh $IMG -c 'java -version'
//   docker run [-]-rm [-]-entrypoint sh $IMG -c 'cat /opt/opennms/karaf-version.txt'
//
// The integration-api range is not observed but declared: it is the feature
// range in karaf-features/src/main/resources/features.xml.
